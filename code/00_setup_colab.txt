# SETUP B — Import everything and TEST that the crystal model works.
# If any import fails here, STOP. Do not run later cells — fix the install first.
import sklearn, xgboost, matminer, pymatgen
import numpy as np, pandas as pd
print('sklearn:', sklearn.__version__)
print('xgboost:', xgboost.__version__)
print('matminer:', matminer.__version__)
# pymatgen has no __version__ in newer versions — use importlib.metadata instead
from importlib.metadata import version as _pkg_version
print('pymatgen:', _pkg_version('pymatgen'))

# The crystal model — 3 imports you will use everywhere:
from pymatgen.core import Structure, Composition, Element
from matminer.datasets import load_dataset
from matminer.featurizers.composition import ElementProperty

# TEST 1: a perovskite formula as a Composition object
comp = Composition('SrTiO3')
print('\nTEST 1 — Composition SrTiO3:', comp)
print('elements:', [str(e) for e in comp.elements])
print('amounts:', {str(e): float(comp[e]) for e in comp.elements})
print('Sr is A-site, Ti is B-site, O is X-site. That is ABX3.')

# TEST 2: Magpie featuriser exists and can be created
feat = ElementProperty.from_preset('magpie')
print('\nTEST 2 — Magpie featuriser ready:', type(feat).__name__)
print('SETUP B done — if you see SrTiO3 above, the crystal model works.')
